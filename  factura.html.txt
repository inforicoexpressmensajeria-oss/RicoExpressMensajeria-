        // Signature Pad
        let signaturePad;
        function initSignaturePad() {
            const canvas = document.getElementById('signaturePad');
            signaturePad = {
                canvas: canvas,
                ctx: canvas.getContext('2d'),
                drawing: false
            };
            
            // Set canvas size
            canvas.width = canvas.offsetWidth;
            canvas.height = 200;
            
            // Style
            signaturePad.ctx.strokeStyle = '#000';
            signaturePad.ctx.lineWidth = 2;
            signaturePad.ctx.lineCap = 'round';
            
            // Events
            canvas.addEventListener('mousedown', startDrawing);
            canvas.addEventListener('mousemove', draw);
            canvas.addEventListener('mouseup', stopDrawing);
            canvas.addEventListener('mouseout', stopDrawing);
            
            // Touch events
            canvas.addEventListener('touchstart', handleTouch);
            canvas.addEventListener('touchmove', handleTouch);
            canvas.addEventListener('touchend', stopDrawing);
        }

        function startDrawing(e) {
            signaturePad.drawing = true;
            const rect = signaturePad.canvas.getBoundingClientRect();
            signaturePad.ctx.beginPath();
            signaturePad.ctx.moveTo(e.clientX - rect.left, e.clientY - rect.top);
        }

        function draw(e) {
            if (!signaturePad.drawing) return;
            const rect = signaturePad.canvas.getBoundingClientRect();
            signaturePad.ctx.lineTo(e.clientX - rect.left, e.clientY - rect.top);
            signaturePad.ctx.stroke();
        }

        function stopDrawing() {
            signaturePad.drawing = false;
        }

        function handleTouch(e) {
            e.preventDefault();
            const touch = e.touches[0];
            const mouseEvent = new MouseEvent(e.type === 'touchstart' ? 'mousedown' : 'mousemove', {
                clientX: touch.clientX,
                clientY: touch.clientY
            });
            signaturePad.canvas.dispatchEvent(mouseEvent);
        }

        function clearSignature() {
            signaturePad.ctx.clearRect(0, 0, signaturePad.canvas.width, signaturePad.canvas.height);
            currentSignature = null;
        }

        function saveSignature() {
            currentSignature = signaturePad.canvas.toDataURL();

        function closeModal() {
            document.getElementById('serviceModal').classList.remove('active');
        }

        function regenerateInvoice(id) {
            const service = services.find(s => s.id === id);
            if (service) {
                currentService = service;
                generateInvoice(service);
                showPage('invoice');
            }
        }

        function updateStatus(id) {
            const service = services.find(s => s.id === id);
            const statuses = ['Pendiente', 'En Camino', 'Entregado'];
            const currentIndex = statuses.indexOf(service.status);
            const nextStatus = statuses[(currentIndex + 1) % statuses.length];
            
            service.status = nextStatus;
            localStorage.setItem('rex_services', JSON.stringify(services));
            closeModal();
            renderHistory();
            alert(`Estado actualizado a: ${nextStatus}`);
        }

        function deleteService(id) {
            if (confirm('¿Está seguro de eliminar este servicio?')) {
                services = services.filter(s => s.id !== id);
                localStorage.setItem('rex_services', JSON.stringify(services));
                closeModal();
                renderHistory();
                updateStats();
            }
        }

        // Export Functions
        function downloadPDF() {
            const element = document.getElementById('invoicePrint');
            if (!element) return;
            
            const opt = {
                margin: 10,
                filename: `Factura_${currentService.guideNumber}.pdf`,
                image: { type: 'jpeg', quality: 0.98 },
                html2canvas: { scale: 2 },
                jsPDF: { unit: 'mm', format: 'letter', orientation: 'portrait' }
            };
            
            html2canvas(element).then(canvas => {
                const imgData = canvas.toDataURL('image/png');
                const { jsPDF } = window.jspdf;
                const pdf = new jsPDF('p', 'mm', 'letter');
                const imgWidth = 190;
                const pageHeight = 277;
                const imgHeight = canvas.height * imgWidth / canvas.width;
                let heightLeft = imgHeight;
                let position = 10;
                
                pdf.addImage(imgData, 'PNG', 10, position, imgWidth, imgHeight);
                heightLeft -= pageHeight;
                
                while (heightLeft >= 0) {
                    position = heightLeft - imgHeight + 10;
                    pdf.addPage();
                    pdf.addImage(imgData, 'PNG', 10, position, imgWidth, imgHeight);
                    heightLeft -= pageHeight;
                }
                
                pdf.save(`Factura_${currentService.guideNumber}.pdf`);
            });
        }

            if (!currentService) return;

                `*Entrega:* ${currentService.deliveryAddress}\n\n` +
                `Gracias por confiar en nosotros! 🚚`;
            
            const url = `https://wa.me/${currentService.clientPhone}?text=${encodeURIComponent(text)}`;
            window.open(url, '_blank');
        }

            const service = services.find(s => s.id === id);
            if (service) {
                currentService = service;
                shareWhatsApp();
            }
        }

        // Utilities
        function formatDate(dateString) {
            if (!dateString) return '-';
            const date = new Date(dateString);
            return date.toLocaleDateString('es-CO', { 
                year: 'numeric', 
                month: 'short', 
                day: 'numeric',
                hour: '2-digit',
                minute: '2-digit'
            });
        }

        function getStatusColor(status) {
            const colors = {
                'Pendiente': 'pending',
                'En Camino': 'transit',
                'Entregado': 'delivered'
            };
            return colors[status] || 'pending';
        }

        function updateStats() {
            const total = services.length;
            const delivered = services.filter(s => s.status === 'Entregado').length;
            const pending = services.filter(s => s.status === 'Pendiente').length;
            const revenue = services.reduce((sum, s) => sum + s.price, 0);
            
            document.getElementById('statTotal').textContent = total;
            document.getElementById('statDelivered').textContent = delivered;
            document.getElementById('statPending').textContent = pending;
            document.getElementById('statRevenue').textContent = '$' + revenue.toLocaleString();
        }
